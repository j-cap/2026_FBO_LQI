\section{Problem Formulation}
\subsection{Heterogeneous clients and local controller design}
Consider clients $i\in\mathcal C$ with true plants
\begin{align}
 x_{i,k+1}&=f_i(x_{i,k},u_{i,k},w_{i,k}),\\
 y_{i,k}&=h_i(x_{i,k},v_{i,k}),
\end{align}
where $f_i$ differs across clients. Each client identifies a compact linear control-design model
\begin{equation}
 x_{k+1}=A(\theta_i)x_k+B(\theta_i)u_k,
 \label{eq:identified_model}
\end{equation}
with estimate $\hat\theta_i$. In the vehicle study, $x=[\beta,r]^\top$. The true evaluation plant includes nonlinear tire saturation, whereas identification and LQI synthesis retain the lower-order linear model.

For yaw-rate tracking, augment the state with integral tracking error
\begin{equation}
 z_{k+1}=z_k+T_s(r_k^{\rm ref}-r_k).
\end{equation}
With $x_k^a=[x_k^\top,z_k]^\top$, parameterize the LQI penalty by $\xi\in\Xi\subset\mathbb R^3$,
\begin{align}
 Q(\xi)&=\operatorname{diag}(10^{\xi_1},10^{\xi_2},10^{\xi_3}),\\
 R&=1.
 \label{eq:lqi_weights}
\end{align}
After fixed physical state scaling, the target-specific feedback gain is
\begin{equation}
 K_i(\xi)=\operatorname{LQI}(\hat A_i,\hat B_i,\hat C_i,Q(\xi),R).
 \label{eq:lqi_gain}
\end{equation}
Thus all clients share the same calibration coordinates, while the resulting gains remain client specific.

For calibration episode $\omega$, applying $K_i(\xi)$ to the true plant gives tracking loss $J_i(\xi;\omega)$ and objective
\begin{equation}
 f_i(\xi)=\mathbb E_{\omega}[J_i(\xi;\omega)],
 \label{eq:client_objective}
\end{equation}
subject to stability, steering-demand, and sideslip constraints. Every client uses the same balanced multi-maneuver task distribution.

\subsection{Mature-fleet transfer objective}
A mature client $j$ provides historical calibration records
\begin{equation}
 \calD_j^{\rm hist}=\{(\xi_{j,m},\tilde y_{j,m})\}_{m=1}^{M_j},
\end{equation}
where $\tilde y_{j,m}$ is its normalized calibration loss. For a new target $i$, its own history is excluded from the source pool. The practical quantity of interest is the number of \emph{new target-local experiments}. Let $J_i^\star$ be an offline oracle used only for evaluation. Define
\begin{equation}
 N_{i,5\%}=\min\{n:J_i^{\rm best}(n)\le1.05J_i^\star\}.
 \label{eq:n5}
\end{equation}
The main question is whether historical fleet data reduce $N_{i,5\%}$ relative to independent BO.

\section{Fleet-Informed Bayesian Optimization}
\subsection{Normalized historical observations}
Historical objectives can have different physical scales. Every source observation is normalized by a fixed client-local baseline controller,
\begin{equation}
 \tilde y_{j,m}=\frac{J_j(\xi_{j,m})}{J_j^{\rm base}}.
 \label{eq:normalization}
\end{equation}
The target uses the same normalization, while infeasible evaluations receive a fixed normalized penalty. The baseline is independent of the oracle used for reporting.

\subsection{Warm-started GP surrogate}
At local evaluation $n$, the target fits a Gaussian-process surrogate to
\begin{equation}
 \calD_i^{(n)}=\calD_i^{\rm local}(n)\cup\bigcup_{j\ne i}\calD_j^{\rm hist}
\end{equation}
and selects the next candidate through expected improvement,
\begin{equation}
 \xi_{i,n+1}\in\arg\max_{\xi\in\Xi}\operatorname{EI}_i(\xi\mid\calD_i^{(n)}).
 \label{eq:ei}
\end{equation}
The target then synthesizes $K_i(\xi_{i,n+1})$ and evaluates it locally. Mature records are static prior information; no new source-client rollout is required.

We compare (i) independent BO, using only target observations; (ii) global fleet warm start, using all mature observations equally; and (iii) dynamics-weighted warm start as a mechanistic ablation.

\subsection{Dynamics-weighted ablation}
For clients $i,j$, define the uncertainty-aware dynamics distance
\begin{equation}
(\distdyn_{ij})^2=(\hat\theta_i-\hat\theta_j)^\top(\hat\Sigma_i+\hat\Sigma_j+\epsilon I)^{-1}(\hat\theta_i-\hat\theta_j),
\end{equation}
and similarity
\begin{equation}
 s_{ij}=\exp\left(-\frac{(\distdyn_{ij})^2}{2\ell_d^2}\right),
\end{equation}
with median-distance bandwidth $\ell_d$. Lower similarity is converted into higher GP observation noise,
\begin{equation}
 \alpha_{ij}=\frac{\alpha_0}{\max(s_{ij},\epsilon_s)}.
\end{equation}
This weighting changes the surrogate and sometimes the acquisition decision, so its usefulness is tested empirically rather than assumed.
